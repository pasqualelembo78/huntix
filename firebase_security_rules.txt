{

    // ── STANZE OUTDOOR MULTIPLAYER ──────────────────────────
    "outdoor_rooms": {
      "$roomCode": {
        ".read": "auth != null",
        // Solo l'host può creare/modificare la stanza
        ".write": "auth != null && (!data.exists() && newData.child('hostId').val() == auth.uid || data.exists() && data.child('hostId').val() == auth.uid)",

        "outdoor_eggs": {
          "$eggId": {
            // Solo l'host può creare/aggiornare/eliminare le uova; i giocatori possono solo segnare "caught"
            ".write": "auth != null && (!data.exists() || newData.child('caught').val() == true) && (data.child('hostId').val() == auth.uid || newData.child('caughtBy').val() == auth.uid)",
            ".validate": "newData.hasChildren(['element', 'rarity', 'position', 'caught']) && (!newData.child('caught').val() || newData.hasChild('caughtBy'))"
          }
        },
        "scores": {
          "$playerId": {
            ".write":  "auth != null && auth.uid == $playerId && newData.child('eggsFound').val() <= 50 && newData.child('totalMs').val() >= 0"
          }
        },
        "players": {
          "$playerId": {
            ".write": "auth != null && auth.uid == $playerId",
            ".validate": "newData.hasChildren(['playerName', 'online'])"
          }
        },
        ".validate": "newData.hasChildren(['hostName', 'roomName', 'createdAt'])"
      }
    },

    // ── STANZE INDOOR MULTIPLAYER ────────────────────────────
    "indoor_rooms": {
      "$roomCode": {
        ".read":  "auth != null",
        // Solo l'host crea la stanza; i giocatori possono aggiornare il proprio stato
        ".write": "auth != null && (!data.exists() && newData.child('meta').child('hostId').val() == auth.uid || data.exists() && (data.child('meta').child('hostId').val() == auth.uid || $roomCode == 'players'))",

        "scores": {
          "$playerId": {
            ".write": "auth != null && auth.uid == $playerId",
            ".validate": "newData.child('eggsFound').val() <= 50 && newData.child('totalMs').val() >= 0"

          }
        },
        "setup": {
          ".write": "auth != null && data.parent().child('meta').child('hostId').val() == auth.uid"

          ".write": "auth != null && data.parent().child('meta').child('hostId').val() == auth.uid"
        },
        "meta": {
          ".write": "auth != null && (!data.exists() && newData.child('hostId').val() == auth.uid || data.exists() && data.child('hostId').val() == auth.uid)"
        }
      }
    },

    // ── CHAT GLOBALE ─────────────────────────────────────────
    "chat_global": {
      "$messageId": {
        ".read":  "auth != null",
        ".write": "auth != null && newData.child('uid').val() == auth.uid",
        ".validate": "newData.hasChildren(['uid', 'name', 'text', 'ts']) && newData.child('text').val().length <= 500"
      }
    },

    // ── CHAT PRIVATA (tra amici) ─────────────────────────────
    "chat_private": {
      "$chatId": {
        ".read":  "auth != null && $chatId.contains(auth.uid)",
        "$messageId": {
          ".write": "auth != null && $chatId.contains(auth.uid) && newData.child('uid').val() == auth.uid",
          ".validate": "newData.hasChildren(['uid', 'name', 'text', 'ts']) && newData.child('text').val().length <= 500"
        }
      }

    "match_rooms": {

        ".write": false

    "$other": {

// ═══════════════════════════════════════════════════════════════
//  FIRESTORE — Production Security Rules (Complete)
//  Huntix — com.intelligame.huntix
//  Incolla in: Firebase Console → Firestore Database → Regole
// ═══════════════════════════════════════════════════════════════

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── HELPER FUNCTIONS ──────────────────────────────────────

    function isAuth() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return isAuth() && request.auth.uid == uid;
    }

    function isValidPlayerWrite() {
      let incoming = request.resource.data;
      let allowed = incoming.keys().hasOnly([
        'displayName', 'avatarUrl', 'statusMessage',
        'preferences', 'city', 'gender', 'birthYear'
      ]);
      let sizeOk = incoming.size() <= 25;
      return allowed && sizeOk;
    }

    function isTeamMember(teamId) {
      return isAuth() && request.auth.uid in
        get(/databases/$(database)/documents/teams/$(teamId)).data.members;
    }

    // ── PROFILI GIOCATORI ──────────────────────────────────────
    match /players/{playerId} {
      // Solo il proprietario legge il proprio profilo completo
      allow read:   if isOwner(playerId);
      // Profilo pubblico leggibile da tutti (nome, avatar, badge)
      allow read:   if isAuth() && request.auth.uid != playerId
        && request.resource == null;  // read-only access handled above
      // Solo il proprietario può modificare, con whitelist campi
      allow write:  if isOwner(playerId) && isValidPlayerWrite();

      match /quests/{questId} {
        allow read, write: if isOwner(playerId);
      }
    }

    // ── ABILITÀ ────────────────────────────────────────────────
    match /abilities/{abilityId} {
      allow read:   if isAuth();
      allow write:  if false;  // Solo Cloud Functions / admin
    }

    // ── PERSONALIZZAZIONE ──────────────────────────────────────
    match /customization/{itemId} {
      allow read:   if isAuth();
      allow write:  if false;
    }

    // ── AVATAR / CONFIG ────────────────────────────────────────
    match /players/{playerId}/avatar/{docId=**} {
      allow read, write: if isOwner(playerId);
    }

    // ── SQUADRE ─────────────────────────────────────────────────
    match /teams/{teamId} {
      allow read:   if isAuth();
      allow create: if isAuth() && request.resource.data.leaderId == request.auth.uid;
      allow update: if isAuth() && (
        request.auth.uid in resource.data.members ||
        request.auth.uid == resource.data.leaderId
      );
      allow delete: if isAuth() && request.auth.uid == resource.data.leaderId;

      match /messages/{msgId} {
        allow read:   if isAuth() && isTeamMember(teamId);
        allow create: if isAuth() && isTeamMember(teamId)
          && request.resource.data.uid == request.auth.uid;
        allow update, delete: if false;
      }
    }

    // ── LIVE EVENTS ─────────────────────────────────────────────
    match /live_events/{eventId} {
      allow read:   if isAuth();
      allow write:  if false;  // Solo Cloud Functions / admin

      match /participants/{uid} {
        allow read:  if isAuth();
        allow write: if isOwner(uid);
      }
    }

    // ── RICHIESTE AMICIZIA ──────────────────────────────────────
    match /friend_requests/{targetUid} {
      allow read: if isOwner(targetUid);

      match /incoming/{senderUid} {
        allow create: if isOwner(senderUid);
        allow delete: if isOwner(targetUid);
        allow read:   if isOwner(targetUid) || isOwner(senderUid);
      }
    }

    // ── LISTA AMICI ─────────────────────────────────────────────
    match /friends/{userId} {
      allow read: if isOwner(userId);

      match /list/{friendId} {
        allow read:   if isOwner(userId);
        // Scrittura bidirezionale controllata
        allow write:  if isAuth() && (
          request.auth.uid == userId ||
          request.auth.uid == friendId
        );
      }
    }

    // ── CODICI REFERRAL ────────────────────────────────────────
    match /referral_codes/{code} {
      allow read:   if isAuth();
      allow create: if isAuth() && request.resource.data.ownerUid == request.auth.uid;
      allow update: if false;
      allow delete: if false;
    }

    // ── SESSIONI DI GIOCO ───────────────────────────────────────
    match /sessions/{sessionId} {
      allow read:   if isAuth() && resource.data.hostUid == request.auth.uid;
      allow create: if isAuth() && request.resource.data.hostUid == request.auth.uid;
      allow update: if isAuth() && resource.data.hostUid == request.auth.uid;
      allow delete: if false;
    }
    match /game_sessions/{sessionId} {
      allow create: if isAuth() && request.resource.data.hostUid == request.auth.uid;
      allow read:   if isAuth() && resource.data.hostUid == request.auth.uid;
      allow update, delete: if false;
    }

    // ── LEADERBOARD ─────────────────────────────────────────────
    match /leaderboard/{playerId} {
      allow read:  if isAuth();
      allow write: if false;  // Solo Cloud Functions
    }

    // ── STATISTICHE SESSIONE ────────────────────────────────────
    match /player_stats/{uid}/sessions/{sessionId} {
      allow read, write: if isOwner(uid);
    }

    // ── UOVA OUTDOOR (spawn sulla mappa) ────────────────────────
    match /world_eggs/{eggId} {
      allow read:   if isAuth();
      allow create: if isAuth();
      // Solo chi cattura l'uovo può aggiornarlo
      allow update: if isAuth()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['caught', 'caughtBy', 'caughtAt'])
        && request.resource.data.caughtBy == request.auth.uid;
      allow delete: if false;
    }

    // ── PALESTRE ────────────────────────────────────────────────
    match /gyms/{gymId} {
      allow read:   if isAuth();
      allow write:  if isAuth() && request.resource.data.size() <= 20;
    }
    match /gym_trainings/{trainId} {
      allow read:   if isAuth();
      allow create: if isAuth() && request.resource.data.userId == request.auth.uid;
      allow update: if isAuth() && request.resource.data.userId == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['score', 'completed', 'updatedAt']);
      allow delete: if false;
    }

    // ── STANZE OUTDOOR MIRROR (Firestore) ──────────────────────
    match /outdoor_rooms/{code} {
      allow read:  if isAuth();
      allow write: if false;  // Solo Cloud Functions
    }

    // ── MINI GAME STATS ────────────────────────────────────────
    match /mini_game_stats/{userId} {
      allow read:  if isAuth();
      allow write: if isOwner(userId) && validMiniGameStats();
    }
    match /mini_game_leaderboards/{gameId}/top100/{rank} {
      allow read:  if isAuth();
      allow write: if false;
    }
    match /mini_game_events/{eventId} {
      allow read:  if isAuth();
      allow write: if false;  // Solo admin / Cloud Functions
    }
    match /mini_game_daily_rewards/{date} {
      allow read:  if isAuth();
      allow write: if false;  // Solo admin
    }

    // ── SCAMBI CREATURE ─────────────────────────────────────────
    match /trade_offers/{offerId} {
      allow read: if isAuth() && (
        resource.data.fromUid == request.auth.uid ||
        resource.data.toUid == request.auth.uid
      );
      allow create: if isAuth()
        && request.resource.data.fromUid == request.auth.uid
        && request.resource.data.status == "pending";
      allow update: if isAuth()
        && resource.data.toUid == request.auth.uid
        && resource.data.status == "pending"
        && request.resource.data.status in ["accepted", "rejected"];
      allow delete: if false;
    }

    // ── ANTI-CHEAT VALIDATION ──────────────────────────────────
    function validMiniGameStats() {
      let data = request.resource.data;
      return data.totalMvcEarned <= 100000
          && data.longestStreak <= 100
          && data.totalGamesPlayed <= 10000;
    }

    // ── ACQUISTI VERIFICATI SERVER-SIDE ────────────────────────
    match /purchases/{purchaseId} {
      allow read:  if auth != null && resource.data.uid == auth.uid;
      allow write: if false;  // Solo Cloud Functions
    }

    // ── REGISTRO ECONOMICO ──────────────────────────────────────
    match /economy_log/{entryId} {
      allow read, write: if false;  // Solo Cloud Functions
    }

    // ── MATCH ROOMS ────────────────────────────────────────────
    match /match_rooms/{roomId} {
      allow read:  if auth != null;
      allow write: if false;  // Solo Cloud Functions
    }

    // ── REFERRAL LOG ──────────────────────────────────────────
    match /referral_log/{entryId} {
      allow read:  if auth != null && resource.data.uid == auth.uid;
      allow write: if false;  // Solo Cloud Functions
    }

    // ── ANALYTICS EVENTS ──────────────────────────────────────
    match /analytics_events/{eventId} {
      allow read, write: if false;  // Solo Cloud Functions
    }

    // ── ADMIN TOKENS ─────────────────────────────────────────
    match /_admins/{docId} {
      allow read, write: if false;  // Solo Cloud Functions
    }

    // ── BLOCCA TUTTO IL RESTO ──────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
