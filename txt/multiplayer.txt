city_players = {

        "vehicle_code": "",               # codice auto se in guida

        "last_update": 1693737600.0,      # epoch timestamp
    },
    ...
}

GRID_SIZE = 100.0  # metri per lato della zona

def zone_key(lat, lon):
    """Converte coordinate geo in chiave zona."""
    # Proiezione semplice: 1 grado lat ≈ 111km, 1 grado lon ≈ 85km (a lat 41°)
    x = int((lon - LON_ORIGIN) * 111000 / GRID_SIZE)
    y = int((lat - LAT_ORIGIN) * 111000 / GRID_SIZE)
    return f"{x}_{y}"

4.2 — Eventi Socket.IO (namespace /city)

Client → Server:
  "city:join"       → { token, lat, lon, heading, skin }
  "city:move"       → { lat, lon, heading, anim_state }
  "city:vehicle"    → { vehicle_code, action: "enter"|"exit"|"park" }
  "city:chat"       → { text }  (messaggio globale opzionale)
  "city:leave"      → {}

Server → Client:
  "city:welcome"    → { player_id, players: [... zona attuale] }
  "city:player_join"→ { player_id, username, level, skin, lat, lon, heading }
  "city:player_move"→ { player_id, lat, lon, heading, anim_state }
  "city:player_leave"→ { player_id }
  "city:vehicle_update" → { player_id, vehicle_code, action }
  "city:chat"       → { player_id, username, text }

4.3 — Logica di zona (subscribe/unsubscribe)

Quando il client si muove e la zona cambia:
  1. Calcola nuova zona: new_zone = zone_key(new_lat, new_lon)
  2. Se new_zone != old_zone:
     a. Unsubscribe dalla vecchia zona
     b. Subscribe alla nuova zona
     c. Ricevi tutti i player nella nuova zona (city:welcome con lista)
     d. Invia city:player_leave ai vecchi vicini
     e. Invia city:player_join ai nuovi vicini

  Zona visibile = 3x3 attorno alla zona del player (9 zone totali)
  Perimeter = zone adiacenti che entrano/escono dalla vista

4.4 — Cleanup automatico

  TIMEOUT = 30 secondi senza heartbeat
  Se last_update > now - TIMEOUT:
    - Rimuovi il player dallo stato
    - Invia city:player_leave a tutti gli iscritti alle sue zone
    - Log "Player {username} timeout - disconnected"

Responsabilità:

Classe:

public class MultiplayerManager : MonoBehaviour
{
    public static MultiplayerManager Instance;

    // Configurazione
    private const float HEARTBEAT_MS = 0.1f;      // 100ms
    private const float INTERPOLATION_SPEED = 8f;  // Lerp speed
    private const string NAMESPACE = "/city";

    // Stato
    private float heartbeatTimer;
    private string myPlayerId;
    private bool connected;
    private Dictionary<string, RemotePlayer> remotePlayers;

    // Riferimenti
    private CitySocketClient socketClient;  // client Socket.IO Unity
    private Transform playerTransform;      // Game.Instance.player.transform

    // API pubblica
    public bool IsConnected => connected;
    public int PlayerCount => remotePlayers.Count;
    public IReadOnlyDictionary<string, RemotePlayer> Others => remotePlayers;

    void Awake()
    {
        Instance = this;
        remotePlayers = new Dictionary<string, RemotePlayer>();
    }

    void Start()
    {
        // Trova il player locale
        playerTransform = Game.Instance.player.transform;
        // Connetti al namespace /city
        ConnectToCitySocket();
    }

    void Update()
    {
        if (!connected) return;

        // Heartbeat: manda posizione ogni 100ms
        heartbeatTimer += Time.deltaTime;
        if (heartbeatTimer >= HEARTBEAT_MS)
        {
            heartbeatTimer = 0f;
            SendPosition();
        }

        // Interpola tutti i remote player verso la loro posizione target
        foreach (var rp in remotePlayers.Values)
        {
            rp.Interpolate(INTERPOLATION_SPEED);
        }
    }

    void SendPosition()
    {
        // Manda lat, lon, heading, anim_state al server
        Vector3 pos = playerTransform.position;
        GeoCoord geo = WorldOrigin.ToGeo(pos);
        float heading = playerTransform.eulerAngles.y;
        string anim = Game.Instance.player.IsMoving ? "walk" : "idle";

        socketClient.Emit("city:move", new {
            lat = geo.lat,
            lon = geo.lng,
            heading = heading,
            anim_state = anim
        });
    }

    // Callback dal Socket.IO client
    void OnPlayerJoin(string playerId, string username, int level,
                      string skin, float lat, float lon, float heading)
    {
        if (playerId == myPlayerId) return;  // ignora te stesso
        if (remotePlayers.ContainsKey(playerId)) return;

        Vector3 worldPos = WorldOrigin.ToWorld(lat, lon);
        RemotePlayer rp = RemotePlayer.Spawn(
            playerId, username, level, skin, worldPos, heading);
        remotePlayers[playerId] = rp;
    }

    void OnPlayerMove(string playerId, float lat, float lon,
                      float heading, string animState)
    {
        if (!remotePlayers.TryGetValue(playerId, out RemotePlayer rp)) return;
        Vector3 targetPos = WorldOrigin.ToWorld(lat, lon);
        rp.SetTarget(targetPos, heading, animState);
    }

    void OnPlayerLeave(string playerId)
    {
        if (remotePlayers.TryGetValue(playerId, out RemotePlayer rp))
        {
            rp.Destroy();
            remotePlayers.Remove(playerId);
        }
    }

    void OnWelcome(string myId, List<PlayerState> players)
    {
        myPlayerId = myId;
        foreach (var p in players)
        {
            OnPlayerJoin(p.id, p.username, p.level, p.skin,
                        p.lat, p.lon, p.heading);
        }
    }
}

5.2 — Nuovo file: Assets/City/Scripts/Multiplayer/RemotePlayer.cs

  - Interpolare posizione/rotazione verso il target

Classe:

public class RemotePlayer : MonoBehaviour
{
    public string PlayerId;
    public string Username;
    public int Level;
    public string Skin;

    // Stato interpolato
    private Vector3 targetPosition;
    private float targetHeading;
    private string animState;

    // Componenti
    private Animator animator;
    private CharacterWalker walker;
    private Transform nameTag;
    private TextMeshPro nameTagTmp;

    // Template: il prefab del player locale (o characterMedium)
    private static GameObject prefab;

    public static RemotePlayer Spawn(string id, string username, int level,
                                     string skin, Vector3 pos, float heading)
    {
        if (prefab == null)
            prefab = Resources.Load<GameObject>("Characters/characterMedium");

        GameObject go = Instantiate(prefab);
        go.name = "RemotePlayer_" + username;
        go.tag = "Untagged";  // NON "Player" — importantissimo

        // Rimuovi PlayerController e CharacterController
        var pc = go.GetComponent<PlayerController>();
        if (pc != null) Destroy(pc);
        var cc = go.GetComponent<CharacterController>();
        if (cc != null) Destroy(cc);

        var rp = go.AddComponent<RemotePlayer>();
        rp.PlayerId = id;
        rp.Username = username;
        rp.Level = level;
        rp.Skin = skin;
        rp.targetPosition = pos;
        rp.targetHeading = heading;

        // Applica skin
        PlayerAppearance.ApplyTo(go, skin);

        // Setup collider kinematico (come NPC)
        var capsule = go.AddComponent<CapsuleCollider>();
        capsule.height = 1.7f;
        capsule.radius = 0.25f;
        capsule.center = new Vector3(0f, 0.85f, 0f);

        var rb = go.AddComponent<Rigidbody>();
        rb.isKinematic = true;
        rb.useGravity = false;

        // Setup animazioni
        go.GetComponentInChildren<Animator>();

        // Setup name tag
        rp.SetupNameTag();

        go.transform.position = pos;
        go.transform.rotation = Quaternion.Euler(0, heading, 0);

        return rp;
    }

    public void SetTarget(Vector3 pos, float heading, string anim)
    {
        targetPosition = pos;
        targetHeading = heading;
        animState = anim;
    }

    public void Interpolate(float speed)
    {
        // Movimento smooth
        transform.position = Vector3.Lerp(
            transform.position, targetPosition,
            speed * Time.deltaTime);

        // Rotazione smooth
        Quaternion targetRot = Quaternion.Euler(0, targetHeading, 0);
        transform.rotation = Quaternion.Slerp(
            transform.rotation, targetRot,
            speed * Time.deltaTime);

        // Animazioni
        bool moving = animState == "walk" || animState == "run";
        if (animator != null)
            animator.SetFloat("Speed", moving ? 1f : 0f);
    }

    void SetupNameTag()
    {
        // Stessa logica di NPCController.EnsureNameTag()
        // ma con nome utente + livello + indicatore "PLAYER"
        var go = new GameObject("NameTag");
        nameTag = go.transform;
        nameTag.SetParent(transform, false);
        nameTag.localPosition = new Vector3(0f, 2.6f, 0f);

        nameTagTmp = go.AddComponent<TextMeshPro>();
        nameTagTmp.fontSize = 2.4f;
        nameTagTmp.alignment = TextAlignmentOptions.Center;
        nameTagTmp.color = new Color(0.4f, 0.8f, 1f);  // azzurro per player
        go.SetActive(false);
    }

    void UpdateNameTag()
    {
        // Mostra/nasconde in base alla distanza dalla camera
        // Stessa logica di NPCController.UpdateNameTag()
        if (nameTag == null) return;
        var cam = Camera.main;
        if (cam == null) { nameTag.gameObject.SetActive(false); return; }

        float distSqr = (cam.transform.position - transform.position).sqrMagnitude;
        bool show = distSqr < 400f;
        nameTag.gameObject.SetActive(show);
        if (!show) return;

        nameTag.rotation = cam.transform.rotation;
        nameTagTmp.text = "🎮 " + Username +
            "\n<size=60%>Lv." + Level + "</size>";
    }

    public void Destroy()
    {
        Destroy(gameObject);
    }
}

5.6 — Filtro di vicinanza (MAX_REMOTE_PLAYERS)

  PROBLEMA: se 50 player entrano nella stessa zona, il telefono non puo'
  renderizzarli tutti (calo frame, memoria, calore).

  SOLUZIONE: il telefono mostra solo i N player piu' vicini. Gli altri
  esistono nel server ma il telefono li ignora (non li renderizza).

  Costanti:

    MAX_REMOTE_PLAYERS = 20    // massimo player visibili sullo schermo
    VISIBLE_RANGE = 200f       // metri: distanza massima per essere visibili
    FILTER_UPDATE_SEC = 0.5f   // ogni 0.5 secondi ricalcola chi mostrare

  Logica in MultiplayerManager.Update():

    void UpdateVisibilityFilter()
    {
        // 1. Prendi tutti i remote player conosciuti
        // 2. Calcola distanza di ciascuno dal player locale
        // 3. Ordina per distanza (piu' vicini prima)
        // 4. I primi MAX_REMOTE_PLAYERS →SetActive(true)
        // 5. Gli altri → SetActive(false)
        // 6. Tutti quelli fuori da VISIBLE_RANGE → SetActive(false)

        var myPos = playerTransform.position;
        var sorted = remotePlayers.Values
            .OrderBy(rp => Vector3.Distance(myPos, rp.transform.position))
            .ToList();

        int shown = 0;
        foreach (var rp in sorted)
        {
            float dist = Vector3.Distance(myPos, rp.transform.position);
            bool shouldBeVisible = shown < MAX_REMOTE_PLAYERS
                                   && dist <= VISIBLE_RANGE;
            rp.gameObject.SetActive(shouldBeVisible);
            if (shouldBeVisible) shown++;
        }
    }

  Chiamata: ogni FILTER_UPDATE_SEC nel Update() del MultiplayerManager.

  PERCHE' FUNZIONA:
  - Il server manda I DATI di tutti i 50 player (lat, lon, heading)
  - Il client li salva tutti nel dictionary remotePlayers
  - Ma SetActive(false) = il GameObject e' spento → nessuna renderizzazione
  - Il telefono non spreca memoria GPU per i player lontani
  - Quando uno si avvicina, rientra nei 20 piu' vicini → SetActive(true)
  - Il passaggio da visibile a invisibile e' graduale (nessun pop-in bruto)

  ECCEZZIONE: i player in veicolo (anim_state == "drive") possono avere
  un raggio di visibilita' maggiore (300m) perché sono piu' facili da
  renderizzare (un solo modello auto, meno poligoni di un personaggio
  con animazioni).

8. FASE 5 — NPC vs REMOTE PLAYER: DISTINZIONE
================================================================================

8.1 — Il problema

  NPC e RemotePlayer sono entrambi modelli Kenney che camminano.
  Il client deve distinguerli per:
    - Non interagire con un RemotePlayer come se fosse un NPC
    - Mostrare nome diverso (nome utente vs nome finto)
    - Non avviare missioni NPC su un RemotePlayer

8.2 — Soluzione

  Tag Unity:
    - NPC: tag "Untagged" (come ora)
    - RemotePlayer: tag "RemotePlayer" (nuovo tag da creare in Unity)

  Layer Unity (opzionale ma consigliato):
    - Layer "NPC" per i pedoni
    - Layer "RemotePlayer" per gli altri giocatori
    - Il raycast del TouchInputHandler ignora RemotePlayer per interazione

  In NPCController.cs:
    - Nessuna modifica necessaria. Il NPC continua a funzionare come prima.
    - Il raycast di touch rileva solo tag "Untagged" o collider NPC.

  In TouchInputHandler.cs (o dove c'è il raycast per tap):
    - Aggiungere filtro: se colpisci un oggetto con tag "RemotePlayer",
      non fare nulla (o apri profilo player opzionale).

8.3 — Name tag differente

  NPC:     nome finto (Marco, Giulia, ecc.) + ruolo IA + livello amicizia
  Player:  nome utente reale + livello + icona 🎮 + colore azzurro

  La distinzione visiva è immediata.

-- Tabella player connessi (stato volatile, pulita periodicamente)
CREATE TABLE city_active_players (
    user_id     INT PRIMARY KEY REFERENCES users(id),
    username    VARCHAR(64) NOT NULL,
    level       INT DEFAULT 1,
    skin        VARCHAR(32) DEFAULT 'humanMaleA',
    lat         DOUBLE PRECISION,
    lon         DOUBLE PRECISION,
    heading     FLOAT DEFAULT 0,
    zone        VARCHAR(16),
    last_update TIMESTAMP DEFAULT NOW(),
    connected_at TIMESTAMP DEFAULT NOW()

================================================================================
12. NOTE DI IMPLEMENTAZIONE
================================================================================

12.1 — Interpolazione

  Il movimento dei remote player NON deve scattare.
  Usare Vector3.Lerp / Quaternion.Slerp con speed ~8f.
  Il target viene aggiornato dal server ogni ~100ms.
  Il Lerp lo rende smooth tra un update e l'altro.

  Se il server non manda update per >2 secondi,
  il remote player si ferma gradualmente (fade out).

12.2 — Orientamento GPS → Unity

  Il server riceve lat/lon (GPS). Il client deve convertirli
  in coordinate Unity (metri). La funzione WorldOrigin.ToWorld()
  esiste già e fa esattamente questo.

  Attenzione: la conversione dipende da WorldOrigin.LAT/LON
  che è il punto di riferimento della mappa (definito in启动).
  Tutti i client devono avere lo stesso WorldOrigin.

12.3 — Banda

  1 player che manda 10 update/sec × 100 byte = ~1 KB/s in upload
  50 player nella stessa zona = ~50 KB/s in download per player
  Questo è accettabile su 4G/5G/WiFi.

12.4 — Sicurezza

  - Il token JWT deve essere verificato ad ogni city:join
  - Le posizioni possono essere spoofate (il client è authoritative
    per la posizione locale). Per ora è accettabile (gioco casual).
  - Anti-cheat futuro: server-side validation della velocità
    (se un player si muove troppo velocemente, è un cheat).

12.5 — Test

  Per testare il multiplayer localmente:
    1. Avviare il backend con city_realtime.py
    2. Aprire 2 emulatori Android (o 1 emulatore + 1 telefono reale)
    3. Entrambi si connettono al server
    4. Verificare che si vedono a vicenda
    5. Muovere uno e verificare che l'altro lo vede muovere
